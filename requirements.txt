# --- DO NOT UPGRADE numpy TO 2.x -------------------------------------------
# numpy>=2 breaks facenet-pytorch, lpips and scikit-image in this stack. The
# insightface package pins numpy>=2, which is why ArcFace is called through
# onnxruntime directly instead (see scripts/eval_arcface.py). Upgrading numpy
# does not raise an error -- it changes results. Keep this pin.
numpy==1.26.4

# --- Deep learning core ----------------------------------------------------
# CUDA 12.1 builds. For these exact wheels:
#   pip install torch==2.5.1+cu121 torchvision==0.20.1+cu121 \
#       --extra-index-url https://download.pytorch.org/whl/cu121
# CPU-only is fine for evaluation; drop the +cu121 suffix and the index URL.
torch==2.5.1+cu121
torchvision==0.20.1+cu121
